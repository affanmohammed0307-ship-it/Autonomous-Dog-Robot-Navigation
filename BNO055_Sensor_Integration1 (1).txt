/**
 * @brief Updates the OLED display with the current status message and live data.
 * The logic prioritizes: Temporary Message > Welcome Message > Live Status.
 */
void updateOLED() {
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setFont(NULL); // Use default font
  
  // --- STATE 1/2: TEMPORARY or WELCOME MESSAGE ---
  if (millis() < oledDisplayTimeout) {
    
    // Welcome message is a special case to display in two lines
    if (oledStatusMessage.equals("WELCOME BY TEAM 5")) {
      display.setTextSize(2);
      display.setCursor(0, 0); // Top line
      display.println("WELCOME");
      display.setCursor(0, 16); // Second line (16 pixels down)
      display.println("TEAM 5");

    } else {
      // Temporary Command (STOPPING, HEADING TO)
      display.setTextSize(2);
      
      // Use custom fitting for the temporary command messages to fit 2 lines
      if (oledStatusMessage.startsWith("GOING:") || oledStatusMessage.equals("STOPPING...")) {
          
          if (oledStatusMessage.startsWith("GOING:")) {
            display.setTextSize(1);
            display.setCursor(0, 0);
            display.println("COMMAND:");
            display.setTextSize(2);
            display.setCursor(0, 16);
            // Print the target name part
            display.println(oledStatusMessage.substring(6));
          } else { // STOPPING...
            display.setTextSize(2);
            // Center single line
            int16_t x1, y1; uint16_t w, h;
            display.getTextBounds(oledStatusMessage, 0, 0, &x1, &y1, &w, &h);
            display.setCursor((SCREEN_WIDTH - w) / 2, (SCREEN_HEIGHT - 16) / 2);
            display.println(oledStatusMessage);
          }

      } else {
          // Default fallback for any other temporary message
          display.setTextSize(1);
          display.setCursor(0, 0);
          display.println("CMD ACTIVE:");
          display.setCursor(0, 16);
          display.println(oledStatusMessage);
      }
    }
    
    display.display();
    return; // Exit here, don't show live status yet
  }
  
  // --- STATE 3: LIVE ROBOT STATUS (3 lines, standard size 1) ---
  
  // If the timeout has passed, reset the status to the default live message base
  if (oledStatusMessage.equals("STOPPING...") || oledStatusMessage.startsWith("GOING:") || oledStatusMessage.equals("WELCOME BY TEAM 5")) {
    oledStatusMessage = "LIVE STATUS"; 
  }
  
  display.setTextSize(1);
  
  // LINE 1 (Y=0): Detection Mode
  display.setCursor(0, 0);
  String modeString = currentMode == LIDAR_MODE ? "Mode: LIDAR" : "Mode: USONIC";
  display.println(modeString);
  
  // LINE 2 (Y=10): Movement Action (Robot State)
  display.setCursor(0, 10);
  String stateString;
  // This logic must map the robot state/direction to a useful action description
  if (currentState == MOVING && currentDirection.startsWith("Forward")) {
      stateString = "Action: FREE FWD";
  } else if (currentState == MOVING && currentDirection.endsWith("Correct")) {
      stateString = "Action: Correcting";
  } else if (currentState == STOPPED) {
      stateString = "Action: STOPPED";
  } else if (currentState == TURNING) {
      stateString = "Action: TURNING";
  } else {
      stateString = "Action: MOVING"; // Catch-all for other MOVING states
  }
  display.println(stateString);

  // LINE 3 (Y=20): Target Direction
  display.setCursor(0, 20);
  display.println("Dir: " + currentDirection);

  display.display();
}

// Ultrasonic reading function
long readDistance(int trigPin, int echoPin) {
  // ... (Ultrasonic logic remains unchanged)
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);    
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 40000);  
  
  if (duration == 0) return 0;  
  
  long distanceCM = (duration * 0.0343) / 2;  
  
  if (distanceCM > 400) return 400;  
  return distanceCM;
}

// Lidar smoothing helper
int smoothedLidarValue(int newVal) {
  const int LIDAR_SMOOTH_WINDOW = 5;
  const int MAX_LIDAR_RANGE_MM = 12000;
  
  lidarSmoothBuffer[lidarSmoothIndex++] = newVal;
  if (lidarSmoothIndex >= LIDAR_SMOOTH_WINDOW) lidarSmoothIndex = 0;
  long sum = 0; int cnt = 0;
  for (int i = 0; i < LIDAR_SMOOTH_WINDOW; i++) {
    if (lidarSmoothBuffer[i] < MAX_LIDAR_RANGE_MM) { sum += lidarSmoothBuffer[i]; cnt++; }
  }
  if (cnt == 0) return MAX_LIDAR_RANGE_MM;
  return (int)(sum / cnt);
}

String getLidarScanJson() {
    // ... (Lidar JSON creation logic remains unchanged)
    const int PACKET_SIZE = 5;
    const int MAX_LIDAR_RANGE_MM = 12000;
    const int MIN_LIDAR_FILTER_MM = 30;
    const int MAX_PACKETS_PER_POLL = 500;

    String json = "{\"scan\":[";
    int pointCount = 0;
    int minDistance = MAX_LIDAR_RANGE_MM;  
    float minAngle = -1.0;  
    
    // Reset Quadrant minimums before new scan
    minLidarFront = MAX_LIDAR_RANGE_MM;
    minLidarBack = MAX_LIDAR_RANGE_MM;
    minLidarLeft = MAX_LIDAR_RANGE_MM;
    minLidarRight = MAX_LIDAR_RANGE_MM;
    
    for(int i = 0; i < MAX_PACKETS_PER_POLL; i++) {
        
        if (lidarSerial.available() < PACKET_SIZE) {  
            break;  
        }
        
        byte peekByte = lidarSerial.peek();
        if (peekByte == 0x00 || peekByte == 0xA5 || peekByte == 0x5A) {
            lidarSerial.read();  
            continue;
        }
        
        // Read and Parse the Packet
        byte packet[PACKET_SIZE];
        lidarSerial.readBytes(packet, PACKET_SIZE);

        int distance = packet[3] | ((packet[4] & 0x7F) << 8);  
        int raw_angle = packet[1] | ((packet[2] & 0x7F) << 8);  
        float angle = (float)raw_angle / 64.0;  

        if (distance > MIN_LIDAR_FILTER_MM && distance < MAX_LIDAR_RANGE_MM && angle >= 0.0 && angle < 360.0) {  
            
            if (distance < minDistance) {
                minDistance = distance;
                minAngle = angle;  
            }

            // Quadrant Minima Calculation (0 degrees = front)
            int roundedAngle = (int)angle;
            if (roundedAngle >= 315 || roundedAngle < 45) { // Front (315-045)
                minLidarFront = min(minLidarFront, distance);
            } else if (roundedAngle >= 45 && roundedAngle < 135) { // Right (045-135)
                minLidarRight = min(minLidarRight, distance);
            } else if (roundedAngle >= 135 && roundedAngle < 225) { // Back (135-225)
                minLidarBack = min(minLidarBack, distance);
            } else if (roundedAngle >= 225 && roundedAngle < 315) { // Left (225-315)
                minLidarLeft = min(minLidarLeft, distance);
            }

            if (pointCount > 0) {
                json += ",";  
            }
            json += "{\"a\":" + String(angle, 1) + ",\"d\":" + String(distance) + "}";
            pointCount++;
        }
        
        // Increased micro delay for stability
        delayMicroseconds(500);  
    }
    
    // Smooth global minDistance
    minDistance = smoothedLidarValue(minDistance);
    globalMinLidarDist = minDistance;  
    globalMinLidarAngle = minAngle;

    json += "]}";

    String finalJson = "{\"min_dist\":" + String(minDistance) + ",\"min_angle\":" + String(minAngle, 1) + "," + json.substring(1);

    return finalJson;
}

void readIMU() {
  if (bno_initialized) {
    // 1. Fused Orientation (Euler Angles - Kept for dashboard H/R/P display)
    sensors_event_t event;
    bno.getEvent(&event);
    currentHeading = event.orientation.x;
    currentRoll = event.orientation.y;
    currentPitch = event.orientation.z;

    // 2. Calibration Status
    bno.getCalibration(&sys_cal, &gyro_cal, &accel_cal, &mag_cal);
    
    // 3. FUSED QUATERNION DATA (NEW - For robust 3D visualization)
    imu::Quaternion quat = bno.getQuat();
    currentQw = quat.w();
    currentQx = quat.x();
    currentQy = quat.y();
    currentQz = quat.z();

    // 4. Raw Acceleration (m/s^2)
    imu::Vector<3> accel_vec = bno.getVector(Adafruit_BNO055::VECTOR_ACCELEROMETER);
    currentAx = accel_vec.x(); currentAy = accel_vec.y(); currentAz = accel_vec.z();
    currentA_Mag = sqrt(currentAx*currentAx + currentAy*currentAy + currentAz*currentAz);

    // 5. Linear Acceleration (m/s^2 - Gravity Removed)
    imu::Vector<3> linear_accel_vec = bno.getVector(Adafruit_BNO055::VECTOR_LINEARACCEL);
    currentLAx = linear_accel_vec.x();  
    currentLAy = linear_accel_vec.y();  
    currentLAz = linear_accel_vec.z();

    // 6. Raw Gyroscope (dps)
    imu::Vector<3> gyro_vec = bno.getVector(Adafruit_BNO055::VECTOR_GYROSCOPE);
    currentGx = gyro_vec.x(); currentGy = gyro_vec.y(); currentGz = gyro_vec.z();
    
    // 7. Raw Magnetometer (uT)
    imu::Vector<3> mag_vec = bno.getVector(Adafruit_BNO055::VECTOR_MAGNETOMETER);
    currentMx = mag_vec.x(); currentMy = mag_vec.y(); currentMz = mag_vec.z();
  }
}

/**
 * @brief Performs basic Dead Reckoning using Linear Acceleration and Heading.
 * NOTE: This is a MINIMAL integration (Euler integration) and will drift quickly.
 * It serves as a proof-of-concept for X/Y tracking.
 */
void updateDeadReckoning() {
    unsigned long now = millis();
    // Only proceed if the update interval has passed
    if (now - lastDRUpdate < DR_UPDATE_INTERVAL_MS) {
        return; 
    }
    
    // Calculate the time delta (dt) in seconds
    float dt = (float)(now - lastDRUpdate) / 1000.0f; 
    lastDRUpdate = now;
    
    // --- 1. Get Acceleration in the Robot's FRAME (already done by BNO055 linear accel) ---
    float accelX = currentLAx; // Forward/Backward acceleration
    float accelY = currentLAy; // Side-to-Side acceleration

    // --- 2. Update Velocity (V = V0 + a*dt) in the Robot's FRAME ---
    // Simple filter: only update if acceleration is above a threshold to ignore noise
    const float ACCEL_THRESHOLD = 0.1; // m/s^2 
    if (abs(accelX) > ACCEL_THRESHOLD) {
      estimatedVelX += accelX * dt;
    } else {
      // Apply simple drag/friction model if no significant acceleration
      estimatedVelX *= 0.95; // Decay
    }

    if (abs(accelY) > ACCEL_THRESHOLD) {
      estimatedVelY += accelY * dt;
    } else {
      estimatedVelY *= 0.95; // Decay
    }
    
    // --- 3. Convert Velocity from Robot's FRAME to GLOBAL (X/Y) FRAME (Rotation) ---
    // Heading (Yaw) is the angle relative to North (0 deg). 
    float headingRad = (currentHeading) * DEG_TO_RAD;
    
    // Simple 2D Rotation: GlobalX = Ax*sin(H) + Ay*cos(H), GlobalY = Ax*cos(H) - Ay*sin(H)
    float cosH = cos(headingRad);
    float sinH = sin(headingRad);
    
    // Global Y (North/South) component: (Forward vel * cosH) - (Sideways vel * sinH)
    float globalVelY = (estimatedVelX * cosH) - (estimatedVelY * sinH);
    
    // Global X (East/West) component: (Forward vel * sinH) + (Sideways vel * cosH)
    float globalVelX = (estimatedVelX * sinH) + (estimatedVelY * cosH);
    
    // --- 4. Update Position (P = P0 + V*dt) in the Global Frame ---
    estimatedPosX += globalVelX * dt; // East/West movement (meters)
    estimatedPosY += globalVelY * dt; // North/South movement (meters)
}

/**
 * @brief Calculates the shortest angle to turn (in degrees) to reach the target heading.
 * @param current: Current Heading (0-360)
 * @param target: Target Heading (0-360)
 * @return float: Positive value for right turn, negative for left turn.
 */
float calculateTurnCorrection(float current, float target) {
    float error = target - current;
    
    // Normalize error to be between -180 and 180 degrees (shortest path)
    if (error > 180.0f) {
        error -= 360.0f;
    } else if (error < -180.0f) {
        error += 360.0f;
    }
    
    return error;
}

void readGPS() {
  while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      if (gps.location.isValid()) {
        currentLat = gps.location.lat();
        currentLng = gps.location.lng();
      }
      if (gps.altitude.isValid()) {
        currentAlt = gps.altitude.meters();
      }
      
      // Update Time and Date (CET = UTC+1)
      if (gps.date.isValid() && gps.time.isValid()) {
        
        // Get the UTC time in seconds since midnight
        long time_sec = (long)gps.time.hour() * 3600L +
                          (long)gps.time.minute() * 60L +
                          (long)gps.time.second();
        
        // Apply the timezone offset (CET = +1 hour)
        time_sec += TIMEZONE_OFFSET_SEC;  

        // Calculate CET time components
        // Handle 24-hour wrap for day change
        int current_hour = (time_sec / 3600L) % 24;
        int current_minute = (time_sec % 3600L) / 60L;
        int current_second = time_sec % 60L;
        
        // Format time string (HH:MM:SS)
        char timeBuffer[9];
        sprintf(timeBuffer, "%02d:%02d:%02d", current_hour, current_minute, current_second);
        currentTimeStr = String(timeBuffer);

        // Format date string (DD/MM/YYYY)
        char dateBuffer[11];
        // Note: Full date roll-over logic is complex, generally best handled by RTOS or relying on next GPS sentence.
        // For simple display, we use the raw date, which is mostly fine unless it's midnight.
        sprintf(dateBuffer, "%02d/%02d/%04d", gps.date.day(), gps.date.month(), gps.date.year());
        currentDateStr = String(dateBuffer);
      } else {
        currentTimeStr = "No Fix";
        currentDateStr = "No Fix";
      }
    }
  }
}

void OnDataRecv(const esp_now_recv_info *info, const uint8_t *data, int len) {
  espMessage_t msg;
  memcpy(&msg, data, sizeof(msg));
  Serial.print("📩 Received command: ");
  Serial.println(msg.cmd);

  if (strcmp(msg.cmd, "STOP") == 0) {
      blinkLED(1);
      oledStatusMessage = "STOPPING...";
      oledDisplayTimeout = millis() + 3000; // Show "STOPPING..." for 3 seconds (2.5)
  } else if (strcmp(msg.cmd, "POST BUILDING") == 0) {
      blinkLED(2);
      // Use "GOING: " prefix to trigger custom 2-line display in updateOLED()
      oledStatusMessage = "GOING: POST BLD"; 
      oledDisplayTimeout = millis() + 3000; // Show this message for 3 seconds (2.3)
  } else if (strcmp(msg.cmd, "H BUILDING") == 0) {
      blinkLED(3);
      // Use "GOING: " prefix to trigger custom 2-line display in updateOLED()
      oledStatusMessage = "GOING: H BLD"; 
      oledDisplayTimeout = millis() + 3000; // Show this message for 3 seconds (2.3)
  }
}

            // Use defaults if data is undefined (robustness fix)

            qx = qx || 0.0;

            const rollRad = Math.atan2(2 * (qw * qx + qy * qz), 1 - 2 * (qx * qx + qy * qy));
            const pitchRad = Math.asin(2 * (qw * qy - qz * qx));

            // Convert to Degrees

            const pitchDeg = pitchRad * 180 / Math.PI;
            let yawDeg = yawRad * 180 / Math.PI;

// === NON-BLOCKING LOOP IMPLEMENTATION & CONTROL LOGIC (MODIFIED) ===
void loop() {
  server.handleClient();
  readIMU();  
  readGPS();  
  
  // NEW: Dead Reckoning Update
  updateDeadReckoning(); 
  
  // 1. Update OLED Display on every loop iteration to manage timeouts
  updateOLED();  

  // 2. Only read sensors when the interval has passed (non-blocking)
  if (millis() - lastUSRead >= US_INTERVAL) {
    lastUSRead = millis();
    readAllUltrasonicSensors(); // Reads all 5 sensors now
    
    // 3. IMPLEMENT CONTROL LOGIC

    // --- Declarations ---
    // Define blockage status for all US sensors (50cm is the Lidar/US boundary)
    bool usFrontClose = (distFront < 50);  
    bool usDeg45Close = (distDeg45 < 50);  
    bool usLeftClose = (distLeft < 50);
    bool usRightClose = (distRight < 50);  
    bool usBackClose = (distBack < 50);
    
    // Check if ALL US sensors are clear (for Lidar mode switch)
    bool allUsClear = !usFrontClose && !usDeg45Close && !usLeftClose && !usRightClose && !usBackClose;

    // Define "very close" for emergency stop (10cm)
    bool frontVeryClose = (distFront <= 10);  
    bool deg45VeryClose = (distDeg45 <= 10);  
    bool leftVeryClose = (distLeft <= 10);
    bool rightVeryClose = (distRight <= 10);  
    bool backVeryClose = (distBack <= 10);
    
    // Count how many front-facing sensors are "very close" (<= 10cm)
    int veryCloseCount = frontVeryClose + deg45VeryClose; 
    
    // Check if ANY sensor (front, sides, back) is very close (<= 10cm)
    bool anyUsVeryClose = frontVeryClose || deg45VeryClose || leftVeryClose || rightVeryClose || backVeryClose;
    // --------------------------------------------------------------------------


    // --- Mode Switching: Lidar -> US ---
    // Mode Switch to Lidar if in FREE_TO_GO state AND all US sensors report a clear path (> 50cm).
    if (currentState == MOVING && currentDirection.startsWith("Forward") && allUsClear) {  
        currentMode = LIDAR_MODE;
    }  
    // Switch to US Mode if Lidar detects an obstacle (50cm or less) OR if US detected something close
    else if (globalMinLidarDist <= 500 || !allUsClear) {  
        currentMode = ULTRASONIC_MODE;
    }
    
    // --- Action Logic: US Sensor Trigger (Only in ULTRASONIC_MODE) ---
    if (currentMode == ULTRASONIC_MODE) {

        // CRITICAL CHANGE 1 (Highest Priority): Emergency U-Turn if 2 or more front-facing sensors are <= 10cm
        if (veryCloseCount >= 2) {
            currentState = STOPPED;
            currentDirection = "Reverse"; // U-Turn
        }
        
        // CRITICAL CHANGE 2 (Second Priority): Stop/Reverse if ANY single sensor (front, sides, back) is very close (<= 10cm)
        else if (anyUsVeryClose) {  
            currentState = STOPPED;
            currentDirection = "Reverse";  
        }

        // --- SIMPLIFIED DIRECTIONAL LOGIC (10cm < distance < 50cm) ---
        
        // Blocked Front/Front-Diagonal: Stop and Reverse
        else if (usFrontClose || usDeg45Close) {
            currentState = STOPPED;
            currentDirection = "Reverse";
        }
        
        // Blocked Back: Stop and Forward
        else if (usBackClose) {
            currentState = STOPPED;
            currentDirection = "Forward";
        }

        // Blocked Left Side: Turn Right (Opposite of Left)
        else if (usLeftClose) {
            currentState = TURNING;
            currentDirection = "Right";  
            // --- NEW IMU CONTROL LOGIC: Set a new target heading for a 45-degree turn right ---
            targetHeading = fmod(currentHeading + 45.0f, 360.0f);
        }
        
        // Blocked Right Side: Turn Left (Opposite of Right)
        else if (usRightClose) {  
            currentState = TURNING;
            currentDirection = "Left";
            // --- NEW IMU CONTROL LOGIC: Set a new target heading for a 45-degree turn left ---
            targetHeading = fmod(currentHeading - 45.0f + 360.0f, 360.0f); // Add 360 for safe modulo with negative result
        }  
        
        // --- FALLBACK / HEADING CORRECTION LOGIC (Only if safe path) ---
        else {
            
            // Calculate the heading error (how far off we are from the targetHeading)
            float headingError = calculateTurnCorrection(currentHeading, targetHeading);
            
            // Check if we are currently turning
            if (currentState == TURNING) {
                
                // Check if the desired turn is complete (e.g., within 5 degrees)
                if (abs(headingError) < 5.0f) { 
                    // Turn is complete. Reset to MOVING and set new targetHeading to current heading
                    currentState = MOVING;
                    currentDirection = "Forward";
                    targetHeading = currentHeading; // Lock onto the new direction
                } else {
                    // Turn is NOT complete. Keep turning in the direction set above (Right or Left).
                }
            }
            
            // Default to MOVING FORWARD if safe and not currently turning
            else if (currentState == MOVING || currentState == STOPPED) {
                
                // Reset to MOVING if we were STOPPED by an old US error that has cleared
                if (currentState == STOPPED) {
                    currentState = MOVING;
                    // Note: currentDirection keeps its value from the last move/turn command
                }
                
                // Check for minor drift correction (always try to maintain targetHeading)
                if (abs(headingError) >= 2.0f) { // If drifted by more than 2 degrees
                    // This is a subtle correction, still considered "MOVING"
                    if (headingError > 0) {
                         currentDirection = "Right-Correct"; 
                    } else {
                         currentDirection = "Left-Correct";
                    }
                } else {
                    // If drift is minimal, ensure the state is fully forward
                    currentDirection = "Forward";
                }
            }
        }
    }
  }

/**
 * @brief Reads BNO055 IMU, updates Heading and Linear Acceleration (crucial for DR).
 */
void readIMU() {
  if (bno_initialized) {
    sensors_event_t event;
    bno.getEvent(&event);
    
    float rawHeading = event.orientation.x; 

    // 1. CRITICAL: Apply Global Offset to make current forward direction 0 degrees (Local North)
    float correctedHeading = rawHeading - globalYawOffset;

    // 2. Normalize corrected heading (0 to 360)
    if (correctedHeading < 0) { correctedHeading += 360.0; }
    else if (correctedHeading >= 360) { correctedHeading -= 360.0; }

    currentHeading = correctedHeading; // This is the value used by DR/Map
    
    // Linear Acceleration (Used for Dead Reckoning)
    imu::Vector<3> linear_accel_vec = bno.getVector(Adafruit_BNO055::VECTOR_LINEARACCEL);
    currentLAx = linear_accel_vec.x();  
    currentLAy = linear_accel_vec.y();  
    
    // Update calibration status (minimal)
    uint8_t gyro_cal, accel_cal, mag_cal;
    bno.getCalibration(&sys_cal, &gyro_cal, &accel_cal, &mag_cal);
  }
}

/**
 * @brief Performs Dead Reckoning using Linear Acceleration and Heading.
 * (Modified for Quadrant Fix and Restored Smooth Damping)
 */
void updateDeadReckoning() {
    if (!bno_initialized) return;

    unsigned long now = millis();
    if (now - lastDRUpdate < DR_UPDATE_INTERVAL_MS) {
        return; 
    }
    
    float dt = (float)(now - lastDRUpdate) / 1000.0f; 
    lastDRUpdate = now;
    
    float accelX = currentLAx; 
    float accelY = currentLAy; 
    
    const float VEL_POS_THRESHOLD = 0.05; // 5 cm/s
    
    // 1. Update velocity (Euler Integration + Damping)
    if (abs(accelX) > ACCEL_THRESHOLD) { 
        estimatedVelX += accelX * dt; 
    } else { 
        estimatedVelX *= VELOCITY_DAMPING; // Milder Damping
    } 
    
    if (abs(accelY) > ACCEL_THRESHOLD) { 
        estimatedVelY += accelY * dt; 
    } else { 
        estimatedVelY *= VELOCITY_DAMPING; // Milder Damping
    } 
    
    // --- DR Stabilization (Velocity and Position Damping) ---
    if (abs(estimatedVelX) < VEL_POS_THRESHOLD && abs(estimatedVelY) < VEL_POS_THRESHOLD) {
        estimatedVelX = 0.0;
        estimatedVelY = 0.0;
        // Light damping/friction for static drift
        estimatedPosX *= STATIC_POS_DAMPING; 
        estimatedPosY *= STATIC_POS_DAMPING;
    }
    
    // 2. Convert robot-frame velocity to global-frame velocity (Rotation) - Uses the corrected currentHeading
    float headingRad = (currentHeading) * DEG_TO_RAD;
    float cosH = cos(headingRad);
    float sinH = sin(headingRad);
    
    // DR Model: Y-axis is Forward (North), X-axis is Right (East)
    float globalVelY = (estimatedVelX * cosH) - (estimatedVelY * sinH); 
    float globalVelX = (estimatedVelX * sinH) + (estimatedVelY * cosH); 
    
    // 3. Update position
    // *** CRITICAL FIX: Negate to correct for 3rd Quadrant issue (KEEP THIS) ***
    estimatedPosX += (-globalVelX) * dt; 
    estimatedPosY += (-globalVelY) * dt; 
}

/**
 * @brief Reads GPS serial data, updates currentLat/Lng.
 */
void readGPS() {
  while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      if (gps.location.isValid()) {
        currentLat = gps.location.lat();
        currentLng = gps.location.lng();
      }
    }
  }
}

void saveConfig() {
    if (preferences.begin(NVS_NAMESPACE, false)) { 
        preferences.putFloat(NVS_ANCHOR_LAT_KEY, anchorLat);
        preferences.putFloat(NVS_ANCHOR_LNG_KEY, anchorLng);
        preferences.putString(NVS_MAP_BLOCKS_KEY, mapBlocksJson);
        preferences.putFloat(NVS_YAW_OFFSET_KEY, globalYawOffset); // Save the offset
        preferences.end();
    }
}

void gpsToLocalXY(float lat, float lng, float anchorLat, float anchorLng, float* x_out, float* y_out) {
    if (!anchorSet) {
        *x_out = 0.0; *y_out = 0.0; return;
    }
    
    float dx_m = TinyGPSPlus::distanceBetween(lat, lng, lat, anchorLng);
    float dy_m = TinyGPSPlus::distanceBetween(lat, lng, anchorLat, lng);

    if (lng < anchorLng) { dx_m = -dx_m; } 
    if (lat < anchorLat) { dy_m = -dy_m; } 
    
    *x_out = dx_m;
    *y_out = dy_m;
}

void handleSetAnchor() {
    // 1. Get current RAW IMU reading
    sensors_event_t event;
    bno.getEvent(&event);

    // 2. CRITICAL FIX: Save the raw IMU value as the new offset (Local North Lock)
    globalYawOffset = event.orientation.x; 

    // 3. Set a dummy Anchor GPS value if none exists yet (required by TinyGPSPlus math).
    if (!anchorSet) {
         anchorLat = 40.7128; // Dummy arbitrary coordinate for math to work
         anchorLng = -74.0060;
         anchorSet = true;
    }

    // 4. Reset DR to (0, 0)
    estimatedPosX = 0.0; 
    estimatedPosY = 0.0;
    estimatedVelX = 0.0; 
    estimatedVelY = 0.0;
    
    saveConfig(); 
    
    server.send(200, "text/plain", "Local Origin (0,0) Set: Yaw Offset Locked to current heading.");
}

void handleNewBlock() {
    StaticJsonDocument<JSON_DOC_SIZE> doc; 
    DeserializationError error = deserializeJson(doc, mapBlocksJson);
    JsonArray array = doc.is<JsonArray>() ? doc.as<JsonArray>() : doc.to<JsonArray>(); 
    
    String newLabel;
    do {
        newLabel = "Block" + String(nextBlockIndex++);
        bool exists = false;
        for (JsonObject block : array) {
            if (block["label"].as<String>().equals(newLabel)) {
                exists = true;
                break;
            }
        }
        if (!exists) break;
    } while(nextBlockIndex < 100); 

    JsonObject newBlock = array.createNestedObject();
    newBlock["label"] = newLabel;
    newBlock.createNestedArray("corners");
    
    mapBlocksJson.clear();
    serializeJson(doc, mapBlocksJson);
    saveConfig(); 
    
    server.send(200, "application/json", "{\"label\":\"" + newLabel + "\"}");
}


void handleAddBlock() {
    if (!anchorSet) { server.send(400, "text/plain", "Error: Set Local Origin (0,0) first."); return; }
    
    float blockX, blockY;
    
    // Determine whether to use high-trust GPS data or low-trust DR position
    bool useGPS = gps.location.isValid() && (currentLat != 0.0 || currentLng != 0.0);

    if (useGPS) {
        gpsToLocalXY(currentLat, currentLng, anchorLat, anchorLng, &blockX, &blockY);
        // Note: For advanced mapping, you would adjust DR values toward GPS, but for now we trust GPS Fix for new points
    } else {
        blockX = estimatedPosX;
        blockY = estimatedPosY;
    }

    String label = server.arg("label");
    if (label == "") { label = "Block"; }

    StaticJsonDocument<JSON_DOC_SIZE> doc; 
    DeserializationError error = deserializeJson(doc, mapBlocksJson);
    JsonArray array = doc.is<JsonArray>() ? doc.as<JsonArray>() : doc.to<JsonArray>();
    
    JsonObject targetBlock = JsonObject();
    for (JsonObject block : array) {
        if (block["label"].as<String>().equals(label)) {
            targetBlock = block;
            break;
        }
    }
    
    if (targetBlock.isNull()) {
        targetBlock = array.createNestedObject();
        targetBlock["label"] = label;
        targetBlock.createNestedArray("corners");
    }
    
    JsonArray corners = targetBlock["corners"].as<JsonArray>();
    JsonObject corner = corners.createNestedObject();
    corner["x"] = blockX;
    corner["y"] = blockY;
    
    mapBlocksJson.clear();
    serializeJson(doc, mapBlocksJson);
    
    saveConfig(); 

    server.send(200, "text/plain", "Corner added to " + label + " at local position (X: " + String(blockX, 2) + ", Y: " + String(blockY, 2) + ").");
}

void handleMapData() {
    
    bool gpsFixed = gps.location.isValid() && (currentLat != 0.0 || currentLng != 0.0);
    
    StaticJsonDocument<JSON_DOC_SIZE> doc;
    
    // DR/IMU State
    doc["DR_X"] = estimatedPosX;
    doc["DR_Y"] = estimatedPosY;
    doc["Heading"] = currentHeading; 
    doc["Cal_S"] = sys_cal; 

    // GPS/Anchor State
    doc["Latitude"] = currentLat;
    doc["Longitude"] = currentLng;
    doc["GPSFixed"] = gpsFixed;
    doc["AnchorSet"] = anchorSet;
    doc["AnchorLat"] = anchorLat;
    doc["AnchorLng"] = anchorLng;
    
    // Block State
    doc["NextBlockIndex"] = nextBlockIndex; 

    // Embed MapBlocks JSON
    DeserializationError err = deserializeJson(doc["MapBlocks"], mapBlocksJson);
    if (err) { doc["MapBlocks"] = JsonArray(); }
    
    String jsonOutput;
    serializeJson(doc, jsonOutput);
    server.send(200, "application/json", jsonOutput);
}

void setup() {
  Serial.begin(115200);  

  // 1. IMU & I2C Setup
  Wire.begin(I2C_SDA_PIN, I2C_SCL_PIN); 
  if(bno.begin()) {
    bno_initialized = true;
    delay(100);    
    // Load offset on boot
    if (preferences.begin(NVS_NAMESPACE, true)) { // Read-only access
        globalYawOffset = preferences.getFloat(NVS_YAW_OFFSET_KEY, 0.0);
        preferences.end();
    }
    Serial.println("✅ BNO055 IMU initialized.");
  } else {
    Serial.println("❌ BNO055 IMU failed. Check power/address.");
  }
  
  // 2. WIFI Setup
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password); 
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.print("\n✅ WiFi Connected. Map URL: http://"); Serial.print(WiFi.localIP()); Serial.println("/map");

  // 3. GPS Setup
  gpsSerial.begin(GPSBaud, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);

  // 4. Load Anchor/Blocks from Preferences (NVS)
  if (preferences.begin(NVS_NAMESPACE, true)) { // Read-only access
      anchorLat = preferences.getFloat(NVS_ANCHOR_LAT_KEY, 0.0);
      anchorLng = preferences.getFloat(NVS_ANCHOR_LNG_KEY, 0.0);
      mapBlocksJson = preferences.getString(NVS_MAP_BLOCKS_KEY, "[]");
      preferences.end();
  }
  anchorSet = (anchorLat != 0.0 || anchorLng != 0.0);
  
  // Determine nextBlockIndex (for UI label suggestions)
  StaticJsonDocument<JSON_DOC_SIZE> doc;
  if (!deserializeJson(doc, mapBlocksJson) && doc.is<JsonArray>()) {
      for (JsonObject block : doc.as<JsonArray>()) {
          String label = block["label"].as<String>();
          if (label.startsWith("Block")) {
              int index = label.substring(5).toInt();
              if (index >= nextBlockIndex) { nextBlockIndex = index + 1; }
          }
      }
  } else { nextBlockIndex = 1; }

  // 5. Web Server Setup
  server.on("/", handleRoot);
  server.on("/data", HTTP_GET, handleData);
  server.on("/map", HTTP_GET, handleMap);          
  server.on("/mapdata", HTTP_GET, handleMapData);  
  server.on("/setanchor", HTTP_GET, handleSetAnchor); 
  server.on("/addblock", HTTP_GET, handleAddBlock); 
  server.on("/newblock", HTTP_GET, handleNewBlock); 
  server.on("/clearblocks", HTTP_GET, handleClearBlocks); 
  server.on("/resetdr", HTTP_GET, handleResetDRPosition); 
  server.begin();
  
  lastDRUpdate = millis();
}

  readGPS();

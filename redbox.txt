void updateFilteredGPS(double rawLat, double rawLng) {
    if (currentLat == 0.0) {
        currentLat = rawLat;
        currentLng = rawLng;
        return;
    }

    // Teleportation Check (15m Jump limit)
    double jumpDist = TinyGPSPlus::distanceBetween(currentLat, currentLng, rawLat, rawLng);
    if (jumpDist > 15.0) return; 

    // Low Pass Filter (30% new, 70% old)
    float alpha = 0.3; 
    currentLat = (currentLat * (1.0 - alpha)) + (rawLat * alpha);
    currentLng = (currentLng * (1.0 - alpha)) + (rawLng * alpha);
}

void updateSmoothedHeading(float rawHeading) {
    // Vector averaging to handle 0-360 wrap-around smoothly
    double rad = rawHeading * DEG_TO_RAD;
    double x = cos(rad);
    double y = sin(rad);

    // Exponential Moving Average on vectors
    float alpha = 0.2; // Smoothing factor
    headingSumX = (headingSumX * (1.0 - alpha)) + (x * alpha);
    headingSumY = (headingSumY * (1.0 - alpha)) + (y * alpha);

    // Convert back to degrees
    currentHeading = atan2(headingSumY, headingSumX) * RAD_TO_DEG;
    if (currentHeading < 0) currentHeading += 360;
}

bool isClusteredThreat(int currentDistance) { 
    lastLidarDistances[lidarBufferIndex] = currentDistance;
    lidarBufferIndex = (lidarBufferIndex + 1) % CLUSTER_N_POINTS;

    int min_dist = 6000;
    int max_dist = 0;
    bool buffer_filled = true;

    for (int i = 0; i < CLUSTER_N_POINTS; i++) {
        if (lastLidarDistances[i] == 0) {
            buffer_filled = false;
            break;  
        }
        if (lastLidarDistances[i] < min_dist) min_dist = lastLidarDistances[i];
        if (lastLidarDistances[i] > max_dist) max_dist = lastLidarDistances[i];
    }
    
    if (!buffer_filled) return false;

    return (max_dist - min_dist) <= CLUSTER_MAX_DIFF_MM;
}

long measureUltrasonic(int trigPin, int echoPin) { 
  digitalWrite(trigPin, LOW); delayMicroseconds(2);
  digitalWrite(trigPin, HIGH); delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  long duration = pulseIn(echoPin, HIGH, 25000);  
  if (duration == 0) return 0;  
  long dist = (duration * 17) / 100;  // mm
  return dist;  
}

// Critical Range Checks
bool isCriticallyBlockedFront(long dist) { 
    return dist > 0 && dist >= CRITICAL_STOP_MIN_MM && dist <= CRITICAL_FRONT_MAX_MM; 
}

bool isCriticallyBlockedSide(long dist) { 
    return dist > 0 && dist >= CRITICAL_STOP_MIN_MM && dist <= CRITICAL_STOP_MAX_MM; 
}

bool isCriticallyBlockedBack(long dist) { 
    return dist > 0 && dist >= CRITICAL_BACK_MIN_MM && dist <= CRITICAL_BACK_MAX_MM; 
}

void calculateChannelNavigation() {
    
    // --- 1. Get Reliable Side Data (Ultrasonic) ---
    long ultraLeft = *sensors[2].lastDistance;  // Left Sensor
    long ultraRight = *sensors[3].lastDistance; // Right Sensor
    
    // Check if ultra reading is valid (50cm to 250cm)
    bool trustLeft = (ultraLeft >= ULTRA_MIN_MM && ultraLeft <= ULTRA_MAX_MM);
    bool trustRight = (ultraRight >= ULTRA_MIN_MM && ultraRight <= ULTRA_MAX_MM);

    // Calculate a safe diagonal distance based on the side ultrasonic
    long safeDiagLeft = (long)(ultraLeft * 1.414);
    long safeDiagRight = (long)(ultraRight * 1.414);


    // --- 2. Process FRONT-LEFT and BACK-LEFT (The "Left Push") ---
    long valFL, valBL;
    
    // Front Left (135 deg)
    long rawFL = (dist_FrontLeft > 0) ? dist_FrontLeft : 6000; 
    if (trustLeft && safeDiagLeft < rawFL) {
        valFL = safeDiagLeft; // Ultrasonic override for mirror
    } else {
        valFL = rawFL;
    }

    // Back Left (225 deg)
    long rawBL = (dist_BackLeft > 0) ? dist_BackLeft : 6000; 
    if (trustLeft && safeDiagLeft < rawBL) {
        valBL = safeDiagLeft; // Ultrasonic override for mirror
    } else {
        valBL = rawBL;
    }


    // --- 3. Process FRONT-RIGHT and BACK-RIGHT (The "Right Push") ---
    long valFR, valBR;
    
    // Front Right (45 deg)
    long rawFR = (dist_FrontRight > 0) ? dist_FrontRight : 6000; 
    if (trustRight && safeDiagRight < rawFR) {
        valFR = safeDiagRight; // Ultrasonic override for mirror
    } else {
        valFR = rawFR;
    }

    // Back Right (315 deg)
    long rawBR = (dist_BackRight > 0) ? dist_BackRight : 6000; 
    if (trustRight && safeDiagRight < rawBR) {
        valBR = safeDiagRight; // Ultrasonic override for mirror
    } else {
        valBR = rawBR;
    }


    // --- 4. Calculate The "Push" (Center the Robot) ---
    long leftSpace = (valFL + valBL) / 2;
    long rightSpace = (valFR + valBR) / 2;

    long difference = leftSpace - rightSpace;
    int deadZone = 200; // 20cm deadzone for stability

    // --- 5. Decide Steering ---
    if (difference > deadZone) {
        channelNavCommand = "NAV: STEER LEFT"; 
    }
    else if (difference < -deadZone) {
        channelNavCommand = "NAV: STEER RIGHT";
    }
    else {
        channelNavCommand = "NAV: STRAIGHT";
    }
}

void checkAndSetAutonomousAction() { 
    long frontDist = *sensors[0].lastDistance; 
    long backDist = *sensors[1].lastDistance;  
    long leftDist = *sensors[2].lastDistance;  
    long rightDist = *sensors[3].lastDistance; 
    
    bool frontBlock = isCriticallyBlockedFront(frontDist);  
    bool backBlock = isCriticallyBlockedBack(backDist);      
    bool leftBlock = isCriticallyBlockedSide(leftDist);      
    bool rightBlock = isCriticallyBlockedSide(rightDist);    
    bool anyCriticalStopBlock = frontBlock || backBlock || leftBlock || rightBlock;
    
    switch (currentState) {
        case STATE_IDLE: {
            if (anyCriticalStopBlock) {
                currentState = STATE_STOP_TIMER;
                actionStartTime = millis();
                stepDirection = "NONE";
            } 
            else {
                // If not blocked, calculate centering logic for display/use
                calculateChannelNavigation();
            }
            break;
        }

        case STATE_STOP_TIMER: {
            if (millis() - actionStartTime >= STOP_DURATION_MS) {
                
                // Re-check sensors
                frontBlock = isCriticallyBlockedFront(*sensors[0].lastDistance);
                backBlock = isCriticallyBlockedBack(*sensors[1].lastDistance);
                leftBlock = isCriticallyBlockedSide(*sensors[2].lastDistance);
                rightBlock = isCriticallyBlockedSide(*sensors[3].lastDistance);
                anyCriticalStopBlock = frontBlock || backBlock || leftBlock || rightBlock;

                if (!anyCriticalStopBlock) {
                    stepDirection = "NONE";
                    currentState = STATE_IDLE;
                }
                else if (backBlock) {
                    stepDirection = "MOVE FORWARD";  
                    currentState = STATE_STEP_ACTION;
                    actionStartTime = millis();  
                } 
                else if (frontBlock) {
                    bool isLeftClear = (*sensors[2].lastDistance == 0 || *sensors[2].lastDistance > CRITICAL_STOP_MAX_MM);
                    bool isRightClear = (*sensors[3].lastDistance == 0 || *sensors[3].lastDistance > CRITICAL_STOP_MAX_MM);
                    
                    if (isLeftClear) {
                        stepDirection = "STEP LEFT";
                    } else if (isRightClear) {
                        stepDirection = "STEP RIGHT";
                    } else {
                        stepDirection = "BLOCKED"; 
                    }
                    
                    currentState = STATE_STEP_ACTION;
                    actionStartTime = millis();  
                } 
                else if (leftBlock) { 
                    stepDirection = "STEP RIGHT";  
                    currentState = STATE_STEP_ACTION;
                    actionStartTime = millis();
                } 
                else if (rightBlock) { 
                    stepDirection = "STEP LEFT";  
                    currentState = STATE_STEP_ACTION;
                    actionStartTime = millis();
                } 
            }
            break;
        }
        
        case STATE_STEP_ACTION: {
            if (millis() - actionStartTime >= STEP_ACTION_DURATION_MS) {
                currentState = STATE_IDLE;
                stepDirection = "NONE";  
            }
            break;
        }
    }
}

void readAndBroadcastLidar() { 
  if (LidarSerialPort.available() >= 47) {
    if (LidarSerialPort.read() == 0x54) {
      if (LidarSerialPort.read() == 0x2C) {
        
        // Reset Diagonals
        dist_FrontRight = 6000; 
        dist_FrontLeft = 6000;  
        dist_BackLeft = 6000;   
        dist_BackRight = 6000;  

        uint8_t buf[45];
        LidarSerialPort.readBytes(buf, 45); 

        uint16_t startAngle = buf[2] | (buf[3] << 8);
        uint16_t endAngle = buf[40] | (buf[41] << 8);
        
        if (endAngle < startAngle) endAngle += 36000;
        float step = (float)(endAngle - startAngle) / 11.0;

        for (int i = 0; i < 12; i++) {
            int base = 4 + (i * 3);
            uint16_t rawDistance = buf[base] | (buf[base+1] << 8);
            uint8_t confidence = buf[base+2];

            float currentAngle = (startAngle + step * i) / 100.0;  
            if (currentAngle >= 360.0) currentAngle -= 360.0;
            
            if (rawDistance > 0 && confidence > 100) {
                
                // Capture Diagonals
                if (currentAngle > 30 && currentAngle < 60) {
                    dist_FrontRight = min(dist_FrontRight, (long)rawDistance);
                }
                if (currentAngle > 120 && currentAngle < 150) {
                    dist_FrontLeft = min(dist_FrontLeft, (long)rawDistance);
                }
                if (currentAngle > 210 && currentAngle < 240) {
                    dist_BackLeft = min(dist_BackLeft, (long)rawDistance);
                }
                if (currentAngle > 300 && currentAngle < 330) {
                    dist_BackRight = min(dist_BackRight, (long)rawDistance);
                }
                
                // Note: Not sending JSON to save bandwidth, but reading data for FSM.
            } 
        }
      } else {
        while(LidarSerialPort.available() > 0) LidarSerialPort.read(); 
      }
    }
  }
}

void broadcastUltraData() { 
    if (millis() - lastUltraScanTime >= ultraScanInterval) {
        UltrasonicSensor& current = sensors[currentUltraSensor];
        long rawDist = measureUltrasonic(current.trigPin, current.echoPin);  
        
        *current.lastDistance = rawDist;
        
        // Note: Not sending JSON to save bandwidth, but reading data for FSM.
        
        currentUltraSensor = (currentUltraSensor + 1) % NUM_SENSORS;
        lastUltraScanTime = millis();
    }
}

void broadcastStatus() { 
    if (millis() - lastStatusBroadcastTime >= STATUS_BROADCAST_INTERVAL) {
        
        StaticJsonDocument<256> doc;
        doc["type"] = "STATUS";
        doc["sats"] = satCount;
        doc["cal"] = String(sysCal) + "/" + String(gyroCal) + "/" + String(magCal);

        // Logic Merge: If FSM is active, show Safety Status. Else show GPS or Channel Status.
        if (currentState == STATE_IDLE) {
            doc["action"] = "FREE TO GO";
            
            // If we have satellites, show GPS. If indoor, show Channel
            if (satCount > 3) {
                 doc["nav"] = navMainStatus; // GPS String
            } else {
                 doc["nav"] = channelNavCommand; // Channel String
            }
            doc["target"] = channelNavCommand;

        } else if (currentState == STATE_STOP_TIMER) {
            doc["action"] = "STOPPED";
            doc["nav"] = "OBSTACLE DETECTED";
            doc["target"] = "WAITING...";
        } else {
            doc["action"] = "AVOIDING";
            doc["nav"] = String(stepDirection);
            doc["target"] = String(stepDirection);
        }

        // --- MODIFIED: Add FSM State to JSON for Webpage visual stop light ---
        doc["fsmState"] = currentState;

        char json[256];
        serializeJson(doc, json);
        webSocket.broadcastTXT(json);

        lastStatusBroadcastTime = millis();
    }
}

// MODIFIED: Simplified to only display the instruction string passed.
void displayInstructionsOnOLED(String instruction) {
    
    // OLED: Ensure display is normal (not inverted)
    display.invertDisplay(false); 
    
    display.clearDisplay();
    display.setTextSize(1);
    display.setTextColor(SSD1306_WHITE); 
    display.setCursor(0, 0);

    // --- NEW: Directly display the received instruction string (e.g., "TURN LEFT") ---
    String displayLine = instruction; 
    
    int maxChars = 21;
    String line1 = "";
    String line2 = "";
    
    // Use the instruction for multi-line logic
    if (displayLine.length() > maxChars) {
        int splitPoint = displayLine.lastIndexOf(' ', maxChars);
        if (splitPoint == -1) splitPoint = maxChars;
        line1 = displayLine.substring(0, splitPoint);
        line2 = displayLine.substring(splitPoint); 
        line2.trim(); 
    } else {
        line1 = displayLine;
    }

    int16_t x1, y1;
    uint16_t w, h;
    display.getTextBounds(line1, 0, 0, &x1, &y1, &w, &h);
    display.setCursor((SCREEN_WIDTH - w) / 2, 8);
    display.println(line1);

    if (line2.length() > 0) {
        display.getTextBounds(line2, 0, 0, &x1, &y1, &w, &h);
        display.setCursor((SCREEN_WIDTH - w) / 2, 20); 
        display.println(line2);
    }
    
    display.display();
}

void handleSetTarget() {
    // Check if we received a Compass Offset Adjustment from UI
    if (server.hasArg("offset")) {
        GLOBAL_COMPASS_OFFSET = server.arg("offset").toInt();
        server.send(200, "text/plain", "OFFSET UPDATED");
        return;
    }

    if (server.hasArg("lat") && server.hasArg("lng")) {
        targetLat = server.arg("lat").toDouble();
        targetLng = server.arg("lng").toDouble();
        activeTargetID = 1; 
        navMainStatus = "GPS: TARGET RECEIVED";
    }
    if (server.hasArg("stop")) {
        activeTargetID = -1;
        navMainStatus = "STOPPED BY USER";
    }
    server.send(200, "text/plain", "OK");
}

// --- NEW HANDLER: Receives the exact instruction string from the webpage ---

    if (server.hasArg("dir")) {
        // Decode the URL-encoded string and store it
        webDirectionInstruction = server.arg("dir");
    }
    server.send(200, "text/plain", "OK");
}

function toXY(lat, lng) {
    let x = -((lng - REF.lng) * Math.cos(REF.lat * Math.PI/180) * SCALE);
    let y = (lat - REF.lat) * SCALE; 
    return {x, y};
}

// ** MAP MATCHING (RESTORED FOR WARNINGS) **
function isPointInPolygon(p, polygon) {
    let inside = false;
    for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
        let xi = polygon[i].lat, yi = polygon[i].lng;
        let xj = polygon[j].lat, yj = polygon[j].lng;
        
        let intersect = ((yi > p.lng) != (yj > p.lng)) &&
            (p.lat < (xj - xi) * (p.lng - yi) / (yj - yi) + xi);
        if (intersect) inside = !inside;
    }
    return inside;
}

function getDist(p1, p2) {
    const R=6371e3; const f1=p1.lat*Math.PI/180; const f2=p2.lat*Math.PI/180;
    const df=(p2.lat-p1.lat)*Math.PI/180; const dl=(p2.lng-p1.lng)*Math.PI/180;
    const a=Math.sin(df/2)*Math.sin(df/2) + Math.cos(f1)*Math.cos(f2)*Math.sin(dl/2)*Math.sin(dl/2);
    return R*2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));
}
function getBearing(p1, p2) {
    const y = Math.sin(p2.lng*Math.PI/180 - p1.lng*Math.PI/180) * Math.cos(p2.lat*Math.PI/180);
    const x = Math.cos(p1.lat*Math.PI/180)*Math.sin(p2.lat*Math.PI/180) -
              Math.sin(p1.lat*Math.PI/180)*Math.cos(p2.lat*Math.PI/180)*Math.cos(p2.lng*Math.PI/180 - p1.lng*Math.PI/180);
    const brng = Math.atan2(y, x) * 180 / Math.PI;
    return (brng + 360) % 360;
}
function intersect(p1,p2,p3,p4) {
    // FIX: ADDED MULTIPLICATION OPERATOR (*)
    const ccw = (a,b,c) => (c.y-a.y)*(b.x-a.x) > (b.y-a.y)*(c.x-a.x); 
    const a = toXY(p1.lat,p1.lng), b = toXY(p2.lat,p2.lng);
    const c = toXY(p3.lat,p3.lng), d = toXY(p4.lat,p4.lng);
    return (ccw(a,c,d) != ccw(b,c,d)) && (ccw(a,b,c) != ccw(a,b,d));
}

function isPathClear(n1, n2) {

        for(let i=0; i<b.pts.length; i++) {
            if(intersect(n1, n2, b.pts[i], b.pts[(i+1)%b.pts.length])) return false;
        }
    }
    return true;
}

// Uses the map path to determine the turn, NOT the robot's heading
function getTurnInstruction(prevWp, currWp, nextWp) {
    if (!prevWp || !currWp || !nextWp) return {text: "STRAIGHT", icon: "&#8593;"};

    let angleIn = getBearing(prevWp, currWp);
    let angleOut = getBearing(currWp, nextWp);
    let turnAngle = angleOut - angleIn;
    
    // Normalize angle to -180 to 180
    if(turnAngle > 180) turnAngle -= 360;
    if(turnAngle < -180) turnAngle += 360;

    let invert = document.getElementById('chkInvert').checked;
    let labelRight = invert ? "LEFT" : "RIGHT";
    let labelLeft = invert ? "RIGHT" : "LEFT";
    let iconRight = invert ? "&#8592;" : "&#8594;"; 
    let iconLeft = invert ? "&#8594;" : "&#8592;";

    if (Math.abs(turnAngle) < 25) {
        return {text: "CONTINUE STRAIGHT", icon: "&#8593;"};
    } else if (turnAngle > 45) {
        return {text: "TURN " + labelRight, icon: iconRight};
    } else if (turnAngle > 10) {
        return {text: "SLIGHT " + labelRight, icon: "&#8599;"};
    } else if (turnAngle < -45) {
        return {text: "TURN " + labelLeft, icon: iconLeft};
    } else if (turnAngle < -10) {
        return {text: "SLIGHT " + labelLeft, icon: "&#8598;"};
    }
    return {text: "STRAIGHT", icon: "&#8593;"};
}

// MODIFIED PATHFINDER FOR SPECIFIC H/POST LOGIC
function calcPath(start, end) {
    let startNode = {id:'S',lat:start.lat,lng:start.lng};
    let endNode = {id:'E',lat:end.lat,lng:end.lng};

    // The sequence of waypoints between POST (ID 2) and H (ID 1)
    // TREAT 1, PASSAGE ENTRY, PASSAGE MIDDLE, PASSAGE EXIT, TREAT 3, TREAT 4, TREAT 6
    // Removed TREAT 2 (11) and TREAT 5 (17)
    const corePathIds = [10, 6, 7, 8, 15, 16, 18, 19];


    let targetPath = [];

    // Case 1: Going to H BUILDING (ID 1)
    if (end.id === 1) {
        // Path: [Core Path] -> H
        targetPath = corePathIds.map(id => WAYPOINTS.find(n => n.id === id)).filter(n => n);
        targetPath.push(endNode); // Add H Building at the end
    }
    // Case 2: Going to POST BUILDING (ID 2)
    else if (end.id === 2) {
        // Path: [Reverse Core Path] -> Post
        let revIds = [...corePathIds].reverse();
        targetPath = revIds.map(id => WAYPOINTS.find(n => n.id === id)).filter(n => n);
        targetPath.push(endNode); // Add Post Building at the end
    }
    
    // If a special path was defined, find the closest start point
    if (targetPath.length > 0) {
        let closestWPIndex = -1;
        let minStartDistance = Infinity;
        
        // Find closest point in the sequence to snap to
        for(let i = 0; i < targetPath.length; i++) {
            let dist = getDist(startNode, targetPath[i]);
            if (dist < minStartDistance) {
                minStartDistance = dist;
                closestWPIndex = i;
            }
        }

        if (closestWPIndex !== -1) {
            let finalPath = [startNode];
            
            // --- MODIFIED LOGIC: START FROM CLOSEST WAYPOINT ---
            // This replaces the aggressive back-step (closestWPIndex - 2)
            let startIndex = closestWPIndex; // Start from the closest waypoint in the sequence
            
            for(let i = startIndex; i < targetPath.length; i++) {
                finalPath.push(targetPath[i]);
            }
            return finalPath;
        }
    }

    // Default Fallback for other targets
    return [startNode, endNode];
}

void setup() {
  Serial.begin(115200);
  
  // Hardware Init
  gpsSerial.begin(GPSBaud, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  LidarSerialPort.begin(LIDAR_BAUDRATE, SERIAL_8N1, LIDAR_RX_PIN, LIDAR_TX_PIN);
  
  Wire.begin();
  if(!bno.begin()) {
      Serial.println("IMU FAIL"); 
  } else {
      bno.setExtCrystalUse(true);
  }

  // --- OLED INIT ---
  if(!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for(;;); 
  }
  display.display();
  delay(100);
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);
  display.println(F("ROBOT BOOTING..."));
  display.display();

  // Pin Modes
  pinMode(TRIG_FRONT, OUTPUT); pinMode(ECHO_FRONT, INPUT); digitalWrite(TRIG_FRONT, LOW);
  pinMode(TRIG_BACK, OUTPUT); pinMode(ECHO_BACK, INPUT); digitalWrite(TRIG_BACK, LOW);
  pinMode(TRIG_LEFT, OUTPUT); pinMode(ECHO_LEFT, INPUT); digitalWrite(TRIG_LEFT, LOW);
  pinMode(TRIG_RIGHT, OUTPUT); pinMode(ECHO_RIGHT, INPUT); digitalWrite(TRIG_RIGHT, LOW);

  // WiFi
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  
  // ** PERFORMANCE FIX: Disable Power Saving to prevent Lag **
  WiFi.setSleep(false);
  
  while (WiFi.status() != WL_CONNECTED) { 
      delay(500); 
      Serial.print("."); 
  }
  Serial.println(WiFi.localIP());
  
  // Servers
  server.on("/", handleRoot);
  server.on("/data", handleMapData);
  server.on("/set_target", handleSetTarget);
  // --- NEW SERVER HANDLER (Setup) ---
  server.on("/set_oled_dir", handleSetOLEDDirection);
  // ----------------------------------
  server.begin();
  webSocket.begin();
}

void loop() {
  server.handleClient();
  webSocket.loop();
  
  // 1. SENSOR UPDATE (GPS)
  while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      satCount = gps.satellites.value();
      if (gps.location.isValid()) {
          updateFilteredGPS(gps.location.lat(), gps.location.lng());
      }
    }
  }
  
  // 2. SENSOR UPDATE (IMU)
  bno.getCalibration(&sysCal, &gyroCal, &accelCal, &magCal);
  sensors_event_t event; 
  bno.getEvent(&event);
  float rawHeading = event.orientation.x + GLOBAL_COMPASS_OFFSET;
  if(rawHeading >= 360) rawHeading -= 360;
  updateSmoothedHeading(rawHeading);
  
  // 3. SENSOR UPDATE (LIDAR/ULTRA & BROADCAST)
  readAndBroadcastLidar();
  broadcastUltraData();
  
  // 4. HYBRID NAVIGATION LOGIC
  String navMainStatus = "BOOTING..."; 
  String oledInstruction = "WAITING..."; 

  
  // --- A. SAFETY OVERRIDE (FSM) ---
  checkAndSetAutonomousAction();

  if (currentState != STATE_IDLE) {
      // --- FSM ACTIVE (AVOIDANCE) ---
      navMainStatus = "SAFETY: " + String(stepDirection);
      oledInstruction = "SAFETY: " + String(stepDirection); 
  } else {
      // --- FSM IDLE (SAFE) ---
      
      // Determine if we should use GPS (Outdoor) or Channel (Indoor)
      bool indoorMode = (satCount <= 3);

      if (!indoorMode && activeTargetID != -1) {
           // GPS Active: USE THE INSTRUCTION SENT FROM THE WEBPAGE UI
           oledInstruction = webDirectionInstruction; // <-- Instruction from the webpage
           
           // Keep the navMainStatus updated for the WS broadcast
           distToTarget = TinyGPSPlus::distanceBetween(currentLat, currentLng, targetLat, targetLng);
           double bearing = TinyGPSPlus::courseTo(currentLat, currentLng, targetLat, targetLng);
           float error = bearing - currentHeading;
           if (error > 180) error -= 360; 
           if (error < -180) error += 360;
           char buf[32];
           snprintf(buf, 32, "GPS: %.0f deg (%.0fm)", error, distToTarget);
           navMainStatus = String(buf);
           
      } else {
           // CHANNEL LOGIC (Fallback or default when no GPS target)
           navMainStatus = channelNavCommand; 
           oledInstruction = channelNavCommand; // Full channel command, e.g. "NAV: STEER LEFT"
      }
  }
  
  // 5. OLEd & BROADCAST STATUS
  // The 'oledInstruction' now holds the webpage's exact text or the FSM/Channel instruction
  displayInstructionsOnOLED(oledInstruction);
  broadcastStatus();
  
  // Increased delay slightly to allow WiFi stack processing
  delay(5);
}

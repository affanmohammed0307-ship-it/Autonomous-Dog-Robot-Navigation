function toXY(lat, lng) {
    let x = -((lng - REF.lng) * Math.cos(REF.lat * Math.PI/180) * SCALE);
    let y = (lat - REF.lat) * SCALE; 
    return {x, y};
}

function getDist(p1, p2) {
    const R=6371e3; const f1=p1.lat*Math.PI/180; const f2=p2.lat*Math.PI/180;
    const df=(p2.lat-p1.lat)*Math.PI/180; const dl=(p2.lng-p1.lng)*Math.PI/180;
    const a=Math.sin(df/2)*Math.sin(df/2) + Math.cos(f1)*Math.cos(f2)*Math.sin(dl/2)*Math.sin(dl/2);
    return R*2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));
}

function getBearing(p1, p2) {
    const lat1 = p1.lat * Math.PI/180;
    const lat2 = p2.lat * Math.PI/180;
    const dlng = (p2.lng - p1.lng) * Math.PI/180;

    const y = Math.sin(dlng) * Math.cos(lat2);
    const x = Math.cos(lat1) * Math.sin(lat2) - Math.sin(lat1) * Math.cos(lat2) * Math.cos(dlng);
    
    let bearing = Math.atan2(y, x) * 180 / Math.PI;
    return (bearing + 360) % 360; 
}

// --- PATHFINDING ALGORITHM (kept for brevity) ---
function intersect(p1,p2,p3,p4) {
    const ccw = (a,b,c) => (c.y-a.y)*(b.x-a.x) > (b.y-a.y)*(c.x-a.x); 
    const a = toXY(p1.lat,p1.lng), b = toXY(p2.lat,p2.lng);
    const c = toXY(p3.lat,p3.lng), d = toXY(p4.lat,p4.lng);
    return (ccw(a,c,d) != ccw(b,c,d)) && (ccw(a,b,c) != ccw(a,b,d));
}

function isPathClear(n1, n2) {
    for(let b of BUILDINGS) {
        for(let i=0; i<b.pts.length; i++) {
            if(intersect(n1, n2, b.pts[i], b.pts[(i+1)%b.pts.length])) return false;
        }
    }
    return true;
}

function calcPath(start, end) {
    let nodes = [...WAYPOINTS, {id:'S',lat:start.lat,lng:start.lng}, {id:'E',lat:end.lat,lng:end.lng}];
    let adj = {}; nodes.forEach(n => adj[n.id] = []);
    
    const LINK_DISTANCE_THRESHOLD = 1000; 
    
    for(let i=0; i<nodes.length; i++) {
        for(let j=i+1; j<nodes.length; j++) {
            if(getDist(nodes[i], nodes[j]) < LINK_DISTANCE_THRESHOLD && isPathClear(nodes[i], nodes[j])) {
                let d = getDist(nodes[i], nodes[j]);
                adj[nodes[i].id].push({n:nodes[j], d:d}); adj[nodes[j].id].push({n:nodes[i], d:d});
            }
        }
    }
    
    let open = [{id:'S', f:0}], cameFrom = {}, gScore = {}, fScore = {};
    nodes.forEach(n => { gScore[n.id] = Infinity; fScore[n.id] = Infinity; });
    gScore['S'] = 0; fScore['S'] = getDist(start, end);
    
    while(open.length > 0) {
        open.sort((a,b) => a.f - b.f);
        let curr = open.shift().id;
        if(curr === 'E') {
            let path = [], temp = 'E';
            while(temp in cameFrom){ path.unshift(nodes.find(n => n.id == temp)); temp = cameFrom[temp]; }
            path.unshift(nodes.find(n => n.id == 'S'));
            return path;
        }
        if(!adj[curr]) continue;
        for(let nbr of adj[curr]) {
            let tg = gScore[curr] + nbr.d;
            if(tg < gScore[nbr.n.id]) {
                cameFrom[nbr.n.id] = curr; gScore[nbr.n.id] = tg; fScore[nbr.n.id] = tg + getDist(nbr.n, end);
                if(!open.some(x => x.id === nbr.n.id)) open.push({id:nbr.n.id, f:fScore[nbr.n.id]});
            }
        }
    }
    return [];
}


// --- NAVIGATION COMMAND LOGIC (MODIFIED FOR ROBOT DOG) ---
const WAYPOINT_TOLERANCE = 5;       // 5m for reaching a waypoint
const FINE_ALIGNMENT_DIST = 1.5;    // Distance for fine alignment commands (Step Left/Right)
const TURN_THRESHOLD_DEG = 20;      // Angle to stop moving and start a major turn
const BACKWARD_THRESHOLD_DEG = 140; // Angle to issue a 'Turn Around' command
const ALIGNMENT_ERROR_DEG = 5;      // Small angle for 'Step' commands

function getNavigationInstruction() {
    const currentPos = {lat: curLat, lng: curLng};
    // Calculate final distance only once
    const distanceToFinal = plannedPath.length > 0 ? getDist(currentPos, plannedPath[plannedPath.length-1]) : 0.0;
    
    // 1. Destination Reached Check
    if (plannedPath.length < 2 || targetWaypointIndex >= plannedPath.length) {
        return {instruction: "DESTINATION REACHED", targetBearing: curHead, dist: 0.0};
    }

    const nextTarget = plannedPath[targetWaypointIndex];
    const distanceToTarget = getDist(currentPos, nextTarget);

    // 2. Waypoint Reached Check & Advance
    if (distanceToTarget < WAYPOINT_TOLERANCE) {
        if(targetWaypointIndex < plannedPath.length - 1) {
            targetWaypointIndex++; 
            const newTarget = plannedPath[targetWaypointIndex];
            const newTargetBearing = getBearing(currentPos, newTarget);
            
            // Recalculate heading error for the new target immediately
            let headingError = newTargetBearing - curHead;
            if (headingError > 180) headingError -= 360;
            if (headingError < -180) headingError += 360;
            
            let turnInstruction;
            if (Math.abs(headingError) > BACKWARD_THRESHOLD_DEG) {
                 turnInstruction = "TURN AROUND (180°)";
            } else if (Math.abs(headingError) > TURN_THRESHOLD_DEG) {
                 // Forces a discrete turn command upon reaching the node
                 turnInstruction = (headingError > 0) ? "TURN RIGHT" : "TURN LEFT";
            } else {
                 turnInstruction = "GO STRAIGHT (ALIGNED)"; // Already aligned for the next leg
            }
            
            // Return the next command right away
            return {instruction: "NEXT WP: " + turnInstruction, targetBearing: newTargetBearing, dist: distanceToFinal};
        } else {
            targetWaypointIndex = plannedPath.length; 
            // FINAL DESTINATION REACHED
            return {instruction: "DESTINATION REACHED", targetBearing: curHead, dist: 0.0};
        }
    }

    // 3. Calculate Required Heading and Error to Current Target
    const targetBearing = getBearing(currentPos, nextTarget);
    let headingError = targetBearing - curHead;

    // Normalize error to [-180, 180]
    if (headingError > 180) headingError -= 360;
    if (headingError < -180) headingError += 360;
    
    let instruction;

    // 4. Fine Alignment/Step Commands (Near the next waypoint)
    if (distanceToTarget < FINE_ALIGNMENT_DIST) {
        if (Math.abs(headingError) > TURN_THRESHOLD_DEG) {
             // If close but misaligned, still force a major turn
             instruction = (headingError > 0) ? "TURN RIGHT" : "TURN LEFT";
        } else if (Math.abs(headingError) > ALIGNMENT_ERROR_DEG) {
            // Small heading error near the point: use STEP
            instruction = (headingError > 0) ? "STEP RIGHT" : "STEP LEFT";
        } else {
            instruction = "GO STRAIGHT (ALIGNED)";
        }
    } 
    // 5. Major Turn Commands (Far from the next waypoint)
    else if (Math.abs(headingError) > BACKWARD_THRESHOLD_DEG) {
        instruction = "TURN AROUND (180°)";
    } else if (Math.abs(headingError) > TURN_THRESHOLD_DEG) {
        instruction = (headingError > 0) ? "TURN RIGHT" : "TURN LEFT";
    }
    // 6. Go Straight Command
    else {
        instruction = "GO STRAIGHT (" + distanceToTarget.toFixed(1) + "m)";
    }

    return {instruction, targetBearing, dist: distanceToFinal};
}

function startNav() {
    let id = document.getElementById('targetSelect').value;
    if(id == -1) return;
    
    let startPoint = (curLat == 0) ? REF : {lat:curLat, lng:curLng}; 
    let serverTargetID = id; 

    if (id === '99') {
        const H_BUILDING_ID = 1;
        const CUSTOM_PATH_IDS = [6, 7, 8, 10, 11, 12, 13, 14, 15, 16, H_BUILDING_ID]; 
        serverTargetID = H_BUILDING_ID; 
        
        let fullCustomPath = [];
        CUSTOM_PATH_IDS.forEach(wpID => {
            const wp = TARGETS.find(x => x.id === wpID) || WAYPOINTS.find(x => x.id === wpID);
            if(wp) fullCustomPath.push(wp);
        });

        if (fullCustomPath.length === 0) {
            alert("Custom Path Error: Waypoints not found.");
            return;
        }

        let closestDist = Infinity;
        let closestIndex = 0; 
        
        for(let i = 0; i < fullCustomPath.length; i++) {
            const dist = getDist(startPoint, fullCustomPath[i]);
            if (dist < closestDist) {
                closestDist = dist;
                closestIndex = i;
            }
        }
        
        if (closestDist < WAYPOINT_TOLERANCE) {
             if (closestIndex < fullCustomPath.length - 1) {
                 closestIndex++; 
             } else {
                 alert("Already at or near the final destination!");
                 return;
             }
        }
        
        const remainingPath = fullCustomPath.slice(closestIndex);
        
        plannedPath = [{id:'S', lat:startPoint.lat, lng:startPoint.lng}, ...remainingPath];
        
        if (plannedPath.length <= 1) {
            alert("Path error after calculating closest point.");
            plannedPath = [];
            return;
        }
    } 
    else {
        let t = TARGETS.find(x => x.id == id);
        if(!t) return;
        
        plannedPath = calcPath(startPoint, t); 
    }
    
    if(plannedPath.length == 0) {
        alert("Path Blocked or Too Far!");
    } else {
        targetWaypointIndex = 1; 
        fetch('/set_target?id=' + serverTargetID); 
    }
}

function stopNav() {
    plannedPath = [];
    targetWaypointIndex = 1;
    fetch('/set_target?id=-1');
    document.getElementById('status-display').innerText = "STOPPED";
}

double distanceMeters(double lat1, double lon1, double lat2, double lon2) {
  double R = 6371e3; 
  double phi1 = lat1 * PI / 180.0;
  double phi2 = lat2 * PI / 180.0;
  double dphi = (lat2 - lat1) * PI / 180.0;
  double dlam = (lon2 - lon1) * PI / 180.0;
  double a = sin(dphi/2) * sin(dphi/2) + cos(phi1) * cos(phi2) * sin(dlam/2) * sin(dlam/2);
  double c = 2 * atan2(sqrt(a), sqrt(1-a));
  return R * c;
}

void metersToDeltaLatLng(double dx, double dy, double currentLat, double& dLat, double& dLng) {
    static const double ONE_DEG_LAT = 111320.0; 
    
    double currentLatRad = currentLat * PI / 180.0;
    double oneDegLng = ONE_DEG_LAT * cos(currentLatRad);

    dLat = dy / ONE_DEG_LAT;  
    dLng = dx / oneDegLng;    
}

void setup() {
  Serial.begin(115200);
  
  // WiFi
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nIP: http://" + WiFi.localIP().toString());
  
  // Hardware
  gpsSerial.begin(GPSBaud, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  if(!bno.begin()) Serial.println("IMU Error"); else bno.setExtCrystalUse(true);

  // Server
  server.on("/", handleMapPage);
  server.on("/data", handleData);
  server.on("/set_target", handleSetTarget);
  server.on("/nav_cmd", handleNavCommand); 
  server.begin();

  // Initialize DR timer
  lastDRTime = millis();
}

void loop() {
  server.handleClient();
  
  // =========================================
  // 1. DEAD RECKONING (IMU Integration)
  // =========================================
  unsigned long now = millis();
  double dt = (now - lastDRTime) / 1000.0; 
  lastDRTime = now;
  
  const double ACCEL_THRESHOLD = 0.15; 
  const double VEL_DECAY = 0.98;       

  if (dt > 0.005) { 
    
    // A. Read IMU
    sensors_event_t event; 
    bno.getEvent(&event);
    currentHeading = event.orientation.x; 
    
    // Get Linear Acceleration Vector (m/s^2)
    imu::Vector<3> linearAccel = bno.getVector(Adafruit_BNO055::VECTOR_LINEARACCEL);
    
    // B. Rotate Acceleration Vector from IMU Frame to World Frame (North/East)
    double headingRad = currentHeading * PI / 180.0;
    double acc_N_raw = (linearAccel.x() * cos(headingRad)) - (linearAccel.y() * sin(headingRad)); 
    double acc_E_raw = (linearAccel.x() * sin(headingRad)) + (linearAccel.y() * cos(headingRad)); 
    
    // C. Apply Thresholding and D. Integrate Velocity
    double acc_magnitude = sqrt(acc_E_raw * acc_E_raw + acc_N_raw * acc_N_raw);

    double acc_N = 0.0;
    double acc_E = 0.0;

    if (acc_magnitude > ACCEL_THRESHOLD) {
        acc_N = acc_N_raw;
        acc_E = acc_E_raw;
    }
    velX += acc_E * dt;
    velY += acc_N * dt;

    // E. Apply Velocity Decay (ZUPT effect)
    if (acc_magnitude <= ACCEL_THRESHOLD) {
      velX *= VEL_DECAY;
      velY *= VEL_DECAY;
      if (abs(velX) < 0.005) velX = 0.0; 
      if (abs(velY) < 0.005) velY = 0.0;
    }

    // F. Integrate Position
    double dx = velX * dt; 
    double dy = velY * dt; 
    
    // G. Convert meters displacement to Lat/Lng difference and H. Update DR position
    double dLat, dLng;
    if(drLat != 0.0) { 
      metersToDeltaLatLng(dx, dy, drLat, dLat, dLng);
      drLat += dLat;
      drLng += dLng;
      
      currentLat = drLat;
      currentLng = drLng;
    }
  }


  // =========================================
  // 2. GPS READING & DR CORRECTION (IMU/GPS CALIBRATION)
  // =========================================
  while (gpsSerial.available() > 0) {
    if (gps.encode(gpsSerial.read())) {
      satCount = gps.satellites.value();

      if (gps.location.isValid() && gps.location.age() < 5000) { 
        double rawLat = gps.location.lat();
        double rawLng = gps.location.lng();

        if (rawLat != 0.0 && rawLng != 0.0) {
          
          // --- DR Initialization & Correction ---
          if (drLat == 0.0) { 
             drLat = rawLat;
             drLng = rawLng;
             currentLat = rawLat;
             currentLng = rawLng;
             velX = 0.0;
             velY = 0.0;
             navStatus = "GPS LOCKED. STANDBY.";
          } else {
             // Subsequent Lock: Correct DR state
             double dist = distanceMeters(drLat, drLng, rawLat, rawLng); 
             
             if (dist > 5.0) { 
                drLat = rawLat;
                drLng = rawLng;
                velX = 0.0; 
                velY = 0.0;
             } else if (dist > 1.0) {
                // Weighted correction (blending GPS with DR) for drift reduction
                const double ALPHA_CORRECTION = 0.1; 
                drLat = drLat * (1.0 - ALPHA_CORRECTION) + rawLat * ALPHA_CORRECTION;
                drLng = drLng * (1.0 - ALPHA_CORRECTION) + rawLng * ALPHA_CORRECTION;
                velX *= 0.8; 
                velY *= 0.8;
             }
             currentLat = drLat;
             currentLng = drLng;
          }
        }
      } else {
         if (satCount < 3) navStatus = "WEAK GPS SIGNAL (DR Active)";
      }
    }
  }

  // --- Fallback if DR has not been initialized by GPS but IMU is working ---
  if(drLat == 0.0 && currentHeading != 0.0) {
     navStatus = "DR STANDBY - NEED GPS LOCK";
  }
}
